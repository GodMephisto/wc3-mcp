// Regenerate src/triggerdata.gen.ts from vendor/TriggerData.txt (the World Editor's GUI
// trigger function registry; vendored copy from War3Net, MIT).
//
// The .wtg format does NOT store how many parameters each GUI function has — the editor
// looks that up in TriggerData.txt. Argument-count rule (matches HiveWE): a value counts as
// an argument if it is non-empty, not a number (numbers are version/flag fields), and not
// the literal "nothing". TriggerCalls lines additionally lead with a return type, which is
// not an argument. Usage: node scripts/gen_triggerdata.mjs
import { readFileSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';

const root = join(dirname(fileURLToPath(import.meta.url)), '..');
const src = readFileSync(join(root, 'vendor', 'TriggerData.txt'), 'utf8').replace(/^﻿/, '');

const isNumber = (s) => /^-?\d+$/.test(s);

const events = {};
const conditions = {};
const actions = {};
const calls = {};
const presets = {};
const SECTIONS = {
  TriggerEvents: events,
  TriggerConditions: conditions,
  TriggerActions: actions,
  TriggerCalls: calls,
  TriggerParams: presets,
};

let section = null;
for (const raw of src.split(/\r?\n/)) {
  const line = raw.trim();
  if (!line || line.startsWith('//')) continue;
  if (line.startsWith('[') && line.endsWith(']')) {
    section = line.slice(1, -1);
    continue;
  }
  const target = SECTIONS[section];
  if (!target) continue;
  let eq = line.indexOf('=');
  if (eq === -1) eq = line.indexOf('-'); // rare malformed TriggerCalls lines use '-'
  if (eq <= 0) continue;
  const key = line.slice(0, eq);
  if (key.startsWith('_')) continue; // _Name_Defaults / _Name_Parameters / ... metadata
  const values = line.slice(eq + 1).split(',').map((v) => v.trim());

  if (section === 'TriggerParams') {
    presets[key] = values[1] ?? ''; // preset name -> variable type
    continue;
  }
  const types = values.filter((v) => v && !isNumber(v) && v !== 'nothing');
  if (section === 'TriggerCalls') {
    calls[key] = { returns: types[0] ?? 'unknown', args: types.slice(1) };
  } else {
    target[key] = types;
  }
}

const counts = `${Object.keys(events).length} events, ${Object.keys(conditions).length} conditions, ${Object.keys(actions).length} actions, ${Object.keys(calls).length} calls, ${Object.keys(presets).length} presets`;

const out = `// AUTO-GENERATED by scripts/gen_triggerdata.mjs from vendor/TriggerData.txt — do not edit.
// GUI trigger function signatures the World Editor defines in UI\\TriggerData.txt: the .wtg
// binary format relies on these argument counts to be parseable at all. ${counts}.

export interface TriggerCallSig {
  returns: string;
  args: string[];
}

export const TRIGGER_EVENTS: Record<string, string[]> = ${JSON.stringify(events)};

export const TRIGGER_CONDITIONS: Record<string, string[]> = ${JSON.stringify(conditions)};

export const TRIGGER_ACTIONS: Record<string, string[]> = ${JSON.stringify(actions)};

export const TRIGGER_CALLS: Record<string, TriggerCallSig> = ${JSON.stringify(calls)};

export const TRIGGER_PRESETS: Record<string, string> = ${JSON.stringify(presets)};
`;

writeFileSync(join(root, 'src', 'triggerdata.gen.ts'), out);
console.log(`src/triggerdata.gen.ts written: ${counts}`);
